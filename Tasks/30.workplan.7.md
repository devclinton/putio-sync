### 2. Create Auth Module in putio-sync-api

```rust
// putio-sync-api/src/auth.rs

use argon2::{Argon2, PasswordHash, PasswordHasher, PasswordVerifier};
use argon2::password_hash::SaltString;
use rand::rngs::OsRng;

pub struct AuthConfig {
    pub username: String,
    pub password_hash: String,
    pub session_timeout: Duration,
    pub secret_key: String,
}

pub struct AuthManager {
    config: AuthConfig,
}

impl AuthManager {
    pub fn new(config: AuthConfig) -> Self {
        Self { config }
    }

    /// Verify username and password
    pub fn verify_credentials(&self, username: &str, password: &str) -> Result<bool> {
        if username != self.config.username {
            return Ok(false);
        }

        let parsed_hash = PasswordHash::new(&self.config.password_hash)?;
        Ok(Argon2::default()
            .verify_password(password.as_bytes(), &parsed_hash)
            .is_ok())
    }

    /// Generate password hash
    pub fn hash_password(password: &str) -> Result<String> {
        let salt = SaltString::generate(&mut OsRng);
        let argon2 = Argon2::default();
        let hash = argon2
            .hash_password(password.as_bytes(), &salt)?
            .to_string();
        Ok(hash)
    }

    /// Generate random password
    pub fn generate_password() -> String {
        use rand::Rng;
        const CHARSET: &[u8] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ\
                                 abcdefghijklmnopqrstuvwxyz\
                                 0123456789!@#$%^&*";
        let mut rng = rand::thread_rng();
        (0..16)
            .map(|_| {
                let idx = rng.gen_range(0..CHARSET.len());
                CHARSET[idx] as char
            })
            .collect()
    }

    /// Generate session token
    pub fn create_session(&self, username: &str) -> Result<SessionToken> {
        use jsonwebtoken::{encode, Header, EncodingKey};

        let claims = Claims {
            sub: username.to_string(),
            exp: (Utc::now() + self.config.session_timeout).timestamp() as usize,
        };

        let token = encode(
            &Header::default(),
            &claims,
            &EncodingKey::from_secret(self.config.secret_key.as_bytes()),
        )?;

        Ok(SessionToken {
            token,
            expires_at: Utc::now() + self.config.session_timeout,
        })
    }

    /// Verify session token
    pub fn verify_session(&self, token: &str) -> Result<Claims> {
        use jsonwebtoken::{decode, DecodingKey, Validation};

        let token_data = decode::<Claims>(
            token,
            &DecodingKey::from_secret(self.config.secret_key.as_bytes()),
            &Validation::default(),
        )?;

        Ok(token_data.claims)
    }
}

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    sub: String,  // username
    exp: usize,   // expiration timestamp
}

pub struct SessionToken {
    pub token: String,
    pub expires_at: DateTime<Utc>,
}
```

```rust
// putio-sync-api/src/routes/auth.rs

use axum::{Json, extract::State};
use serde::{Deserialize, Serialize};

#[derive(Deserialize)]
pub struct LoginRequest {
    username: String,
    password: String,
}

#[derive(Serialize)]
pub struct LoginResponse {
    token: String,
    expires_at: String,
}

pub async fn login(
    State(state): State<Arc<ApiState>>,
    Json(req): Json<LoginRequest>,
) -> Result<Json<LoginResponse>, ApiError> {
    // Verify credentials
    if !state.auth.verify_credentials(&req.username, &req.password)? {
        return Err(ApiError::Unauthorized("Invalid credentials".to_string()));
    }

    // Create session
    let session = state.auth.create_session(&req.username)?;

    Ok(Json(LoginResponse {
        token: session.token,
        expires_at: session.expires_at.to_rfc3339(),
    }))
}

pub async fn logout(
    State(state): State<Arc<ApiState>>,
    // Extract token and invalidate
) -> Result<(), ApiError> {
    // Implement token invalidation if needed
    Ok(())
}
```

### 4. Add Auth Middleware

```rust
// putio-sync-api/src/middleware/auth.rs

use axum::{
    extract::{Request, State},
    http::{HeaderMap, StatusCode},
    middleware::Next,
    response::Response,
};

pub async fn require_auth(
    State(state): State<Arc<ApiState>>,
    headers: HeaderMap,
    request: Request,
    next: Next,
) -> Result<Response, StatusCode> {
    // Extract token from Authorization header
    let token = headers
        .get("Authorization")
        .and_then(|v| v.to_str().ok())
        .and_then(|v| v.strip_prefix("Bearer "))
        .ok_or(StatusCode::UNAUTHORIZED)?;

    // Verify token
    state.auth
        .verify_session(token)
        .map_err(|_| StatusCode::UNAUTHORIZED)?;

    Ok(next.run(request).await)
}
```

### 5. First-Run Password Generation

```rust
// putio-sync-api/src/init.rs

pub async fn initialize_auth(config: &mut Config) -> Result<()> {
    let mut config_modified = false;

    // Generate username if not set
    if config.api.username.is_empty() {
        config.api.username = "admin".to_string();
        config_modified = true;
        println!("✓ Set default username: admin");
    }

    // Generate password if not set
    if config.api.password_hash.is_empty() {
        let password = AuthManager::generate_password();
        let hash = AuthManager::hash_password(&password)?;
        config.api.password_hash = hash;
        config_modified = true;

        println!("\n╔══════════════════════════════════════════╗");
        println!("║   API Credentials Generated              ║");
        println!("╠══════════════════════════════════════════╣");
        println!("║ Username: {:<30} ║", config.api.username);
        println!("║ Password: {:<30} ║", password);
        println!("╠══════════════════════════════════════════╣");
        println!("║ IMPORTANT: Save these credentials!       ║");
        println!("║ This password will not be shown again.   ║");
        println!("╚══════════════════════════════════════════╝\n");
    }

    // Generate secret key if not set
    if config.api.secret_key.is_empty() {
        use rand::Rng;
        let secret: String = rand::thread_rng()
            .sample_iter(&rand::distributions::Alphanumeric)
            .take(64)
            .map(char::from)
            .collect();
        config.api.secret_key = secret;
        config_modified = true;
        println!("✓ Generated API secret key");
    }

    // Save config if modified
    if config_modified {
        config.save()?;
        println!("✓ Configuration saved to {}", config.path.display());
    }

    Ok(())
}
```

```rust
// putio-sync-api/src/lib.rs

pub fn create_app(state: Arc<ApiState>) -> Router {
    Router::new()
        // Public routes (no auth required)
        .route("/auth/login", post(auth::login))
        .route("/oauth/callback", get(oauth::callback))  // OAuth exempt

        // Protected routes (auth required)
        .route("/putio/files", get(putio::files::list))
        .route("/putio/files/:id", delete(putio::files::delete))
        .route("/local/files", get(local::files::list))
        .route("/settings", get(settings::get).put(settings::update))
        .layer(middleware::from_fn_with_state(
            state.clone(),
            auth_middleware::require_auth,
        ))

        .with_state(state)
}
```

### 7. Add Password Change Endpoint

```rust
#[derive(Deserialize)]
pub struct ChangePasswordRequest {
    old_password: String,
    new_password: String,
}

pub async fn change_password(
    State(state): State<Arc<ApiState>>,
    claims: Claims,  // From auth middleware
    Json(req): Json<ChangePasswordRequest>,
) -> Result<(), ApiError> {
    // Verify old password
    if !state.auth.verify_credentials(&claims.sub, &req.old_password)? {
        return Err(ApiError::Unauthorized("Invalid password".to_string()));
    }

    // Hash new password
    let new_hash = AuthManager::hash_password(&req.new_password)?;

    // Update config
    let mut config = state.config.write().await;
    config.api.password_hash = new_hash;
    config.save()?;

    Ok(())
}
```

## Security Considerations

1. **Password Storage**
   - Use Argon2 for password hashing
   - Never store plaintext passwords
   - Use cryptographically secure random generation

2. **Session Management**
   - Use JWT with expiration
   - Implement token refresh if needed
   - Consider session invalidation on password change

3. **HTTPS Requirement**
   - Document that API should be behind HTTPS in production
   - Consider adding HTTPS redirect in middleware

4. **Rate Limiting**
   - Add rate limiting to login endpoint
   - Prevent brute force attacks
